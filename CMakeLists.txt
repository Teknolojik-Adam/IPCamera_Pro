cmake_minimum_required(VERSION 3.16)
project(IPCameraProLinux VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

add_library(ipcam_linux SHARED src/ipcam_linux.cpp src/ipcam_pattern.cpp)
target_include_directories(ipcam_linux PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(ipcam_linux PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(ipcam_linux PROPERTIES VERSION 1.0.0 SOVERSION 0)

add_executable(ipcam-cli cli/main.cpp)
target_include_directories(ipcam-cli PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(ipcam-cli PRIVATE ipcam_linux)
target_compile_options(ipcam-cli PRIVATE -Wall -Wextra -Wpedantic)
